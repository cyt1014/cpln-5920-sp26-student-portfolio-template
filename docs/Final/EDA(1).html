library(broom)

score_model <- function(actual, predicted) {
  tibble(
    MAE = mean(abs(actual - predicted), na.rm = TRUE),
    RMSE = sqrt(mean((actual - predicted)^2, na.rm = TRUE)),
    Correlation = cor(actual, predicted, use = "complete.obs")
  )
}

classification_scores <- function(data, truth, pred) {
  tp <- sum(data[[truth]] == 1 & data[[pred]] == 1, na.rm = TRUE)
  fp <- sum(data[[truth]] == 0 & data[[pred]] == 1, na.rm = TRUE)
  fn <- sum(data[[truth]] == 1 & data[[pred]] == 0, na.rm = TRUE)
  tn <- sum(data[[truth]] == 0 & data[[pred]] == 0, na.rm = TRUE)

  tibble(
    Accuracy = (tp + tn) / (tp + tn + fp + fn),
    Precision = ifelse(tp + fp == 0, NA, tp / (tp + fp)),
    Recall = ifelse(tp + fn == 0, NA, tp / (tp + fn)),
    F1 = ifelse(
      (tp + fp) == 0 | (tp + fn) == 0,
      NA_real_,
      2 * ((tp / (tp + fp)) * (tp / (tp + fn))) / ((tp / (tp + fp)) + (tp / (tp + fn)))
    )
  )
}

evaluate_models <- function(data, split_name, model_specs, baseline_var) {
  bind_rows(lapply(names(model_specs), function(model_name) {
    spec <- model_specs[[model_name]]

    predicted <- if (!is.null(spec$type) && spec$type == "baseline") {
      pmax(data[[baseline_var]], 0)
    } else {
      predict(spec$model, newdata = data, type = "response")
    }

    predicted <- pmax(predicted, 0)

    score_model(data$filings_next, predicted) %>%
      mutate(model = model_name, split = split_name)
  }))
}

monthly_model_data <- evict %>%
  ungroup() %>%
  arrange(GEOID, month_date) %>%
  group_by(GEOID) %>%
  mutate(
    filings_next = lead(filings_2020, 1),
    filings_next_per_1k_renters = lead(filings_per_1k_renters, 1),
    lag_2m = lag(filings_2020, 2),
    lag_12m = lag(filings_2020, 12),
    rolling_mean_3m = (lag_1m + lag_2m + lag_3m) / 3
  ) %>%
  ungroup() %>%
  mutate(
    racial_majority = factor(racial_majority),
    month_factor = factor(month_num, levels = 1:12, labels = month.abb)
  ) %>%
  filter(
    !is.na(filings_next),
    !is.na(lag_1m),
    !is.na(lag_2m),
    !is.na(lag_3m),
    !is.na(lag_12m),
    !is.na(renters),
    !is.na(median_gross_rent),
    !is.na(median_household_income),
    !is.na(rent_burden_pct),
    renters > 0
  )

monthly_split_lookup <- tibble(
  month_date = sort(unique(monthly_model_data$month_date))

  mutate(
    data_split = case_when(

monthly_model_data <- monthly_model_data %>%

quarterly_model_data <- evict %>%
  ungroup() %>%
  mutate(
    quarter_date = floor_date(month_date, unit = "quarter"),
    quarter_label = paste0(year(quarter_date), " Q", quarter(quarter_date))
  ) %>%
  group_by(GEOID, quarter_date, quarter_label, racial_majority) %>%
  summarise(
    current_filings = sum(filings_2020, na.rm = TRUE),
    filings_avg = first(filings_avg),
    filings_avg_prepandemic_baseline = first(filings_avg_prepandemic_baseline),
    renters = first(renters),
    median_gross_rent = first(median_gross_rent),
    median_household_income = first(median_household_income),
    rent_burden_pct = first(rent_burden_pct),
    months_in_quarter = n(),
    .groups = "drop"
  ) %>%
  filter(months_in_quarter == 3) %>%
  arrange(GEOID, quarter_date) %>%
  group_by(GEOID) %>%
  mutate(
    filings_next = lead(current_filings, 1),
    filings_next_per_1k_renters = lead((current_filings / renters) * 1000, 1),
    lag_1q = lag(current_filings, 1),
    lag_2q = lag(current_filings, 2),
    change_1q = current_filings - lag_1q,
    rolling_mean_2q = (lag_1q + lag_2q) / 2
  ) %>%
  ungroup() %>%
  mutate(
    racial_majority = factor(racial_majority),
    quarter_factor = factor(quarter(quarter_date), levels = 1:4, labels = c("Q1", "Q2", "Q3", "Q4"))
  ) %>%
  filter(
    !is.na(filings_next),
    !is.na(lag_1q),
    !is.na(lag_2q),
    !is.na(renters),
    !is.na(median_gross_rent),
    !is.na(median_household_income),
    !is.na(rent_burden_pct),
    renters > 0
  )

The model comparison shows that monthly forecasting performs better than quarterly forecasting for predicting future eviction filings. The strongest predictive model is the enhanced monthly baseline, which combines last month’s filings, filings from the same month in the previous year, and a three-month rolling average. This suggests that eviction filings are driven primarily by short-term persistence and recurring seasonal patterns. In contrast, the richer negative binomial models with rent, income, demographic, and seasonal controls do not improve out-of-sample prediction accuracy, although they remain useful for interpreting structural differences in eviction risk across tracts.
