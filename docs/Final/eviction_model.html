Introduction

This report builds a monthly tract-level prediction workflow for Philadelphia eviction filings. The modeling goal is twofold:

  1. Predict next month’s filing count for each census tract.
  2. Flag which tracts are likely to enter a high-risk state.

Following the logic from the EDA, I use monthly data as the main unit of analysis because it preserves short-term dynamics and supports early-warning forecasting.

evict_panel <- evict %>%
  arrange(GEOID, month_date) %>%
  group_by(GEOID) %>%
  mutate(
    filings_rate_1k = (filings_2020 / renters) * 1000,
    filings_next = lead(filings_2020, 1),
    filings_next_rate_1k = lead(filings_rate_1k, 1),
    lag_1m = lag(filings_2020, 1),
    lag_2m = lag(filings_2020, 2),
    lag_3m = lag(filings_2020, 3),
    change_1m = filings_2020 - lag_1m,
    change_3m = filings_2020 - lag_3m,

model_formula <- filings_next ~
  lag_1m +
  lag_3m +
  change_1m +
  rolling_mean_3m +

score_predictions <- function(data, split_name) {
  baseline_scored <- data %>%
    mutate(pred_next = pmax(lag_1m, 0)) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Baseline: lag_1m", split = split_name)

  poisson_scored <- data %>%
    mutate(pred_next = predict(poisson_model, newdata = data, type = "response")) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Poisson", split = split_name)

  nb_scored <- data %>%
    mutate(pred_next = predict(nb_model, newdata = data, type = "response")) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Negative Binomial", split = split_name)

  bind_rows(baseline_scored, poisson_scored, nb_scored) %>%
    dplyr::select(split, model, everything())
}

```{r setup}
score_count_model <- function(data, actual, predicted) {
  tibble(
    MAE = mean(abs(data[[actual]] - data[[predicted]]), na.rm = TRUE),
    RMSE = sqrt(mean((data[[actual]] - data[[predicted]])^2, na.rm = TRUE)),
    Correlation = cor(data[[actual]], data[[predicted]], use = "complete.obs")
  )
}

score_classification <- function(data, truth, pred) {
  tp <- sum(data[[truth]] == 1 & data[[pred]] == 1, na.rm = TRUE)
  fp <- sum(data[[truth]] == 0 & data[[pred]] == 1, na.rm = TRUE)
  fn <- sum(data[[truth]] == 1 & data[[pred]] == 0, na.rm = TRUE)
  tn <- sum(data[[truth]] == 0 & data[[pred]] == 0, na.rm = TRUE)

  precision <- ifelse(tp + fp == 0, NA, tp / (tp + fp))
  recall <- ifelse(tp + fn == 0, NA, tp / (tp + fn))
  accuracy <- (tp + tn) / (tp + tn + fp + fn)
  f1 <- ifelse(is.na(precision) | is.na(recall) | precision + recall == 0,
               NA,
               2 * precision * recall / (precision + recall))

  tibble(
    Accuracy = accuracy,
    Precision = precision,
    Recall = recall,
    F1 = f1
  )
}
```

```{r score-models}
score_predictions <- function(data, split_name) {
  baseline_scored <- data %>%
    mutate(pred_next = pmax(lag_1m, 0)) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Baseline: lag_1m", split = split_name)

  poisson_scored <- data %>%
    mutate(pred_next = predict(poisson_model, newdata = data, type = "response")) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Poisson", split = split_name)

  nb_scored <- data %>%
    mutate(pred_next = predict(nb_model, newdata = data, type = "response")) %>%
    score_count_model("filings_next", "pred_next") %>%
    mutate(model = "Negative Binomial", split = split_name)

  bind_rows(baseline_scored, poisson_scored, nb_scored) %>%
    dplyr::select(split, model, everything())
}
```

# Conclusion

This workflow turns the EDA into a tract-month forecasting pipeline. The model uses recent filing dynamics, historical tract conditions, seasonality, and renter exposure to predict next-month filings, then converts those predictions into a high-risk screening rule. In a policy setting, the final output can be used to prioritize outreach, rental assistance, or legal support in tracts where predicted filing rates are expected to spike.
